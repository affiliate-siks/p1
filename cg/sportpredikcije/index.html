/** * ADMIRALBET SPORT KAMPANJE — BACKEND ZA PREDIKCIJE * --------------------------------------------------- * Ovaj skript ide u Google Apps Script projekat "vezan" (bound) za Google Sheet. * Prati SETUP.md za korak-po-korak uputstvo kako ga instalirati. * * Podržane kampanje (svaka ima svoj tab/list u Sheet-u, kreira se automatski): * - "LigePetice" * - "Evroliga" * * Endpoints: * POST ? -> snima novu predikciju * body (JSON, kao tekst): { campaign, playerId, kolo, picks } * * GET ?action=lookup&campaign=LigePetice&playerId=12345678&kolo=1 * -> vraća predikciju(e) tog igrača (opciono filtrirano po kolu) * * GET ?action=leaderboard&campaign=LigePetice&kolo=1 * -> vraća sve redove za to kolo (za izračun rang liste na frontend strani) */ const SHEET_NAMES = ['LigePetice', 'Evroliga']; function doPost(e) { try { const data = JSON.parse(e.postData.contents); const campaign = data.campaign; if (SHEET_NAMES.indexOf(campaign) === -1) { return jsonOut({ ok: false, error: 'Nepoznata kampanja: ' + campaign }); } const sheet = getOrCreateSheet(campaign); const timestamp = new Date(); // Provjera: da li igrač već ima predikciju za ovo kolo (1 ID = 1 predikcija po kolu) const existing = findRow(sheet, data.playerId, data.kolo); if (existing) { return jsonOut({ ok: false, error: 'Predikcija za ovo kolo je već predata za ovaj ID.' }); } sheet.appendRow([ timestamp, data.playerId, data.kolo, JSON.stringify(data.picks), '', // Bodovi — popunjava se ručno/naknadno nakon settlovanja mečeva ]); return jsonOut({ ok: true, message: 'Predikcija sačuvana.' }); } catch (err) { return jsonOut({ ok: false, error: String(err) }); } } function doGet(e) { const action = e.parameter.action; const campaign = e.parameter.campaign; if (SHEET_NAMES.indexOf(campaign) === -1) { return jsonOut({ ok: false, error: 'Nepoznata kampanja: ' + campaign }); } const sheet = getOrCreateSheet(campaign); if (action === 'lookup') { const playerId = e.parameter.playerId; const kolo = e.parameter.kolo || null; const rows = getAllRows(sheet).filter(r => String(r.playerId) === String(playerId) && (!kolo || String(r.kolo) === String(kolo)) ); return jsonOut({ ok: true, results: rows }); } if (action === 'leaderboard') { const kolo = e.parameter.kolo || null; const rows = getAllRows(sheet).filter(r => !kolo || String(r.kolo) === String(kolo)); return jsonOut({ ok: true, results: rows }); } return jsonOut({ ok: false, error: 'Nepoznata akcija. Koristi ?action=lookup ili ?action=leaderboard' }); } // ---------- helpers ---------- function getOrCreateSheet(name) { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(name); if (!sheet) { sheet = ss.insertSheet(name); sheet.appendRow(['Timestamp', 'PlayerID', 'Kolo', 'Predikcije (JSON)', 'Bodovi']); sheet.setFrozenRows(1); } return sheet; } function findRow(sheet, playerId, kolo) { const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (String(data[i][1]) === String(playerId) && String(data[i][2]) === String(kolo)) { return data[i]; } } return null; } function getAllRows(sheet) { const data = sheet.getDataRange().getValues(); const rows = []; for (let i = 1; i < data.length; i++) { rows.push({ timestamp: data[i][0], playerId: data[i][1], kolo: data[i][2], picks: safeParse(data[i][3]), bodovi: data[i][4], }); } return rows; } function safeParse(str) { try { return JSON.parse(str); } catch (e) { return str; } } function jsonOut(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
